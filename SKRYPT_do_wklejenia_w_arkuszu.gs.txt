 *   2. Usuń całą domyślną zawartość edytora i wklej w to miejsce TEN plik.
 *   3. W linii KLUCZ poniżej wpisz dowolny własny ciąg znaków zamiast
 *      "zmien-to-haslo" — to prosty zamek, żeby adresu nie dało się
 *      przypadkiem odgadnąć. Zapamiętaj go, będzie potrzebny w telefonie.
 *   4. Zapisz (ikona dyskietki).
 *   5. Wdróż → Nowe wdrożenie → (koło zębate) → Aplikacja internetowa.

 *      skrypt na Twoim koncie, zezwól (może pojawić się ostrzeżenie

 *   7. Skopiuj adres "Aplikacja internetowa" (kończy się na /exec).

 *
 * SPRAWDZENIE: wklej gotowy adres (z ?key=...) w Safari. Powinieneś

 *
 * AKTUALIZACJA W PRZYSZŁOŚCI: jeśli kiedyś zmienisz coś w tym skrypcie,

var KLUCZ = 'zmien-to-haslo';

function doGet(e) {
  var podanyKlucz = (e && e.parameter && e.parameter.key) || '';
  if (KLUCZ && podanyKlucz !== KLUCZ) {
    return odpowiedz_({ blad: 'Nieprawidłowy lub brakujący klucz dostępu (?key=...).' });
  }
  try {
    var skoroszyt = SpreadsheetApp.getActiveSpreadsheet();
    var arkusze = [];
    var listaArkuszy = skoroszyt.getSheets();
    for (var i = 0; i < listaArkuszy.length; i++) {
      var sh = listaArkuszy[i];
      var zakres = sh.getDataRange();
      if (zakres.getNumRows() < 1 || zakres.getNumColumns() < 1) continue;
      arkusze.push({
        nazwa: sh.getName(),
        // getDisplayValues() zwraca to, co widać na ekranie (same teksty),
        // dokładnie tak jak eksport CSV — bez tego liczby i daty
        // przychodziłyby jako inne typy danych i psuły rozpoznawanie.
        rows: zakres.getDisplayValues()
      });
    }
    return odpowiedz_({ arkusze: arkusze });
  } catch (blad) {
    return odpowiedz_({ blad: 'Błąd skryptu: ' + String(blad) });
  }
}

function odpowiedz_(dane) {
  return ContentService
    .createTextOutput(JSON.stringify(dane))
    .setMimeType(ContentService.MimeType.JSON);
}
